--HANDLERS--
-- one frame of every scene object with gCullMode = pMode: "lines hidden md5-of-the-points"
function Frame pMode
   global gCullMode, gSceneObjects
   local tBuf, tEntry, tHex
   put pMode into gCullMode
   Render.BeginFrame
   repeat for each line tEntry in gSceneObjects
      RenderMeshInto word 1 of tEntry, word 2 of tEntry, "g", tBuf, true
   end repeat
   get binaryDecode("H*", md5Digest(tBuf["g"]), tHex)
   return Engine.Info("edges") && Engine.Info("hidden") && tHex
end Frame
function NLines pMode
   return word 1 of Frame(pMode)
end NLines
function NHidden pMode
   return word 2 of Frame(pMode)
end NHidden
-- an empty scene, the camera at 0,0,0 looking along +z, not turned
command Fresh

end Fresh
-- true when every tested face's plane has the mesh's middle (0,0,0) on its inner side: normals point out
function NormalsOut pName
   global gMeshCache
   local c
   if gMeshCache[pName]["cullBuilt"] is not true then Mesh.CullBuild pName
   if gMeshCache[pName]["cfCount"] < 1 then return "no faces"
   repeat with c = 1 to gMeshCache[pName]["cfCount"]
      if gMeshCache[pName]["cd"][c] <= 0 then return false
   end repeat
   return true
end NormalsOut
-- the box with every face wound the other way round
function InsideOutBox pW, pH, pD
   local tM, tFace, tNew, tOut, i
   put MeshBox(pW, pH, pD) into tM
   repeat for each line tFace in tM["faces"]
      put empty into tNew
      repeat with i = the number of words of tFace down to 1
         put word i of tFace & space after tNew
      end repeat
      put word 1 to -1 of tNew & return after tOut
   end repeat
   put line 1 to -1 of tOut into tM["faces"]
   return tM
end InsideOutBox
